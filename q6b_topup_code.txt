
    Top-up codes grant N additional scan credits when redeemed.
    Unlike GRAIL- codes (which are tied to subscriptions and
    validated on every scan), TOPUP- codes are one-shot: they
    are redeemed exactly once via /api/redeem-topup, at which
    point credits are written to the redeeming user's modal.Dict
    scan counter and the code is marked spent.

    Args:
        email: purchaser's email from Stripe checkout
        credits: number of scan credits to grant on redemption (default 75)
        payment_intent_id: Stripe payment_intent ID for traceability
    """
    import random, smtplib, ssl
    from datetime import datetime
    from email.mime.text import MIMEText
    from email.mime.multipart import MIMEMultipart

    # Generate TOPUP-XXXX-XXXX code (same charset as GRAIL- codes)
    chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'
    part1 = ''.join(random.choices(chars, k=4))
    part2 = ''.join(random.choices(chars, k=4))
    code = f"TOPUP-{part1}-{part2}"

    # Write redemption record to subscriptions.json
    # Distinct entry shape from subscription codes:
    #   - type: "topup" (vs absent / implicit "subscription" for GRAIL-)
    #   - credits_total / credits_remaining track redemption state
    #   - status: "unredeemed" â†’ "redeemed" on first /api/redeem-topup hit
    subs = _load_subs()
    subs[code] = {
        "type":               "topup",
        "email":              email,
        "credits_total":      credits,
        "credits_remaining":  credits,
        "stripe_payment_intent_id": payment_intent_id,
        "created_at":         datetime.utcnow().isoformat(),
        "status":             "unredeemed",
        "redeemed_at":        None,
        "redeemed_by_device": None,
        "redeemed_by_fp":     None,
    }
    _save_subs(subs)

    # Send email â€” same SMTP path as _issue_new_code
    try:
        smtp_host = os.environ["SMTP_HOST"]
        smtp_port = int(os.environ["SMTP_PORT"])
        smtp_user = os.environ["SMTP_USER"]
        smtp_pass = os.environ["SMTP_PASS"]
        from_address = os.environ.get("SMTP_FROM_NOREPLY", smtp_user)

        msg = MIMEMultipart("alternative")
        msg["Subject"] = f"Your GrailSweep top-up: {credits} scans"
        msg["From"] = from_address
        msg["To"] = email

        body_text = f"""Thanks for your GrailSweep top-up purchase!

Your top-up code: {code}

This code grants you {credits} additional card scans on top of your free monthly allowance. Top-up scans never expire â€” use them whenever you need them.

To redeem your code:
1. Go to https://grailsweep.com/match
2. Enter the code in the "Redeem top-up" field
3. Your {credits} extra scans will be added immediately

If you have any trouble redeeming, contact us at support@grailsweep.com.

Happy collecting!
The GrailSweep team
"""

        body_html = f"""<html><body>
<p>Thanks for your GrailSweep top-up purchase!</p>
<p><strong>Your top-up code:</strong> <code style="font-size:1.2em;background:#f0f0f0;padding:4px 8px;border-radius:4px;">{code}</code></p>
<p>This code grants you <strong>{credits} additional card scans</strong> on top of your free monthly allowance. Top-up scans never expire â€” use them whenever you need them.</p>
<p><strong>To redeem your code:</strong></p>
<ol>
  <li>Go to <a href="https://grailsweep.com/match">grailsweep.com/match</a></li>
  <li>Enter the code in the "Redeem top-up" field</li>
  <li>Your {credits} extra scans will be added immediately</li>
</ol>
<p>If you have any trouble redeeming, contact us at <a href="mailto:support@grailsweep.com">support@grailsweep.com</a>.</p>
<p>Happy collecting!<br>The GrailSweep team</p>
</body></html>"""

        msg.attach(MIMEText(body_text, "plain"))
        msg.attach(MIMEText(body_html, "html"))

        context = ssl.create_default_context()
        with smtplib.SMTP_SSL(smtp_host, smtp_port, context=context) as server:
            server.login(smtp_user, smtp_pass)
            server.sendmail(from_address, email, msg.as_string())

        print(f"[TOPUP] Issued code {code} to {email} ({credits} credits)", flush=True)
    except Exception as e:
        print(f"[TOPUP EMAIL ERROR] {e}", flush=True)

        # doesn't lose the code. Manual recovery possible via direct
        # subscriptions.json lookup.
